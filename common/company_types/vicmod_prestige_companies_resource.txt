# ============================================================================
# A. Копии ванильных basic companies для товаров без prestige good.
#
# Аудит выполнен по vanilla 00_goods.txt и 00_prestige_goods.txt.
# Для каждой компании ниже сохранены без изменений:
# - icon / background / category;
# - building_types и extension_building_types;
# - possible и ai_will_do;
# - prosperity_modifier.
#
# Меняется только назначение possible_prestige_goods и его trigger.
# services / transportation / electricity здесь отсутствуют: в vanilla
# 99_basic_companies.txt нет basic company, владеющей Urban Center, Railway
# или Power Plant соответственно. Создавать ложную "копию" с другими правами
# означало бы нарушить ТЗ.
# ============================================================================

# Копия company_basic_forestry -> prestige good для wood
company_basic_vicmod_wood = {
	icon = "gfx/interface/icons/company_icons/basic_forestry.dds"
	background = "gfx/interface/icons/company_icons/company_backgrounds/comp_illu_farm_corn.dds"

	category = aristocrat_owned

	flavored_company = no

	uses_dynamic_naming = yes

	dynamic_company_type_names = {
		"dynamic_company_type_company"
		"dynamic_company_type_consortium"
		"dynamic_company_type_corporation"
		"dynamic_company_type_conglomerate"
		"dynamic_company_type_syndicate"
		"dynamic_company_type_combine"
		"dynamic_company_type_guild"
	}

	building_types = {
		building_logging_camp
		building_rubber_plantation
	}

	extension_building_types = {
		building_tooling_workshop
	}

	possible_prestige_goods = {
		prestige_good_vicmod_generic_wood
	}

	# У новых prestige goods нет ванильного JE-variable, поэтому доступ открыт сразу.
	prestige_goods_trigger = {
		always = yes
	}

	possible = {
		any_scope_state = {
			any_scope_building = {
				OR = {
					is_building_type = building_logging_camp
					is_building_type = building_rubber_plantation
				}
				level >= 5
			}
		}
	}

	prosperity_modifier = {
		state_infrastructure_from_population_add = 0.7
		state_infrastructure_from_population_max_add = 15
	}

	ai_will_do = {
		any_scope_state = {
			any_scope_building = {
				OR = {
					is_building_type = building_logging_camp
					is_building_type = building_rubber_plantation
				}
				level >= 3
			}
		}
	}
}
